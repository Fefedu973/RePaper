find_package(ZLIB REQUIRED)
set(PDFIO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../external/pdfio)
file(GLOB PDFIO_SOURCES ${PDFIO_ROOT}/pdfio-*.c)
add_library(paper-pdfio STATIC ${PDFIO_SOURCES} ${PDFIO_ROOT}/ttf.c)
set_target_properties(paper-pdfio PROPERTIES AUTOMOC OFF)
target_include_directories(paper-pdfio PUBLIC ${PDFIO_ROOT})
target_compile_definitions(paper-pdfio PRIVATE HAVE_TIMEGM=1 HAVE_TM_GMTOFF=1)
target_link_libraries(paper-pdfio PUBLIC ZLIB::ZLIB m)
add_library(paper-format STATIC SceneWriter.cpp Bundle.cpp NativeImport.cpp)
target_include_directories(paper-format PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(paper-format PUBLIC Qt6::Core Qt6::Gui paper-pdfio ZLIB::ZLIB)
add_library(paper-bridge-runtime STATIC BridgeRuntime.cpp NativeDocumentsClient.cpp)
target_include_directories(paper-bridge-runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(paper-bridge-runtime PUBLIC Qt6::Core Qt6::Network)
add_executable(paper-bridge main.cpp BridgeService.cpp)
target_link_libraries(paper-bridge PRIVATE paper-format paper-bridge-runtime repaper Qt6::Sql Qt6::Network)
set_target_properties(paper-bridge PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS paper-bridge RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(paper-format-tests tests/test_format.cpp)
  target_link_libraries(paper-format-tests PRIVATE paper-format Qt6::Test)
  add_test(NAME paper-format-tests COMMAND paper-format-tests)
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME paper-bridge-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py --binary $<TARGET_FILE:paper-bridge>)
    add_executable(paper-bridge-runtime-probe tests/RuntimeProbe.cpp BridgeService.cpp)
    target_link_libraries(paper-bridge-runtime-probe PRIVATE paper-bridge-runtime paper-format repaper Qt6::Sql)
    add_test(NAME paper-bridge-runtime-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_integration.py --binary $<TARGET_FILE:paper-bridge> --probe $<TARGET_FILE:paper-bridge-runtime-probe>)
  endif()
endif()
