set(XOVI_SOURCE_DIR "" CACHE PATH "Read-only XOVI checkout pinned to 2b99649f5e4fd6288be7792a8570bd16418adb70")
option(REPAPER_BUILD_XOVI_PACKAGE "Build exact-target Xochitl toolbar and native editing module" OFF)
set(SCENE_ASSISTANT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.tools/xovi-scene-assistant" CACHE PATH "Scene Assistant pinned source")
if(REPAPER_BUILD_XOVI_PACKAGE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND git -C ${XOVI_SOURCE_DIR} rev-parse HEAD OUTPUT_VARIABLE XOVI_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE XOVI_STATUS)
  if(NOT XOVI_STATUS EQUAL 0 OR NOT XOVI_COMMIT STREQUAL "2b99649f5e4fd6288be7792a8570bd16418adb70")
    message(FATAL_ERROR "XOVI_SOURCE_DIR must name the documented exact checkout")
  endif()
  execute_process(COMMAND git -C ${SCENE_ASSISTANT_SOURCE_DIR} rev-parse HEAD OUTPUT_VARIABLE SCENE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE SCENE_STATUS)
  if(NOT SCENE_STATUS EQUAL 0 OR NOT SCENE_COMMIT STREQUAL "8afbac01ca7816f9aa22a5897a3f2133b0b0d9d7")
    message(FATAL_ERROR "SCENE_ASSISTANT_SOURCE_DIR must name the documented exact checkout")
  endif()
  # Reuse only Line geometry. SceneLineItem's guessed header/factory and the
  # mutating ensureVtable helper are never compiled or loaded.
  target_sources(repaper-editor-native PRIVATE ${SCENE_ASSISTANT_SOURCE_DIR}/rm_Line.cpp)
  target_include_directories(repaper-editor-native PRIVATE ${SCENE_ASSISTANT_SOURCE_DIR})
  target_compile_definitions(repaper-editor-native PRIVATE REPAPER_WITH_NATIVE_ABI=1)
  target_compile_features(repaper-editor-native PRIVATE cxx_std_20)
  # No resource initializer may run during dlopen before the XOVI constructor.
  # Embed RCC binaries and register them explicitly after installing the QMD.
  configure_file(native/editor.qmd ${CMAKE_CURRENT_BINARY_DIR}/editor.qmd COPYONLY)
  file(GLOB PAPER_UI_RESOURCE_DEPENDENCIES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/ui/qml/*")
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/editor_panels.rcc
    COMMAND Qt6::rcc --binary ${CMAKE_CURRENT_SOURCE_DIR}/editor_panels.qrc -o ${CMAKE_CURRENT_BINARY_DIR}/editor_panels.rcc
    DEPENDS editor_panels.qrc qml/NativeQuickColors.qml qml/NativeStencilQuickBar.qml qml/EditorSpinBox.qml qml/ReInkSidebar.qml qml/NativeEditorPalette.qml qml/StrokeColorPicker.qml qml/StencilPreview.qml qml/StencilConfiguration.qml ${PAPER_UI_RESOURCE_DEPENDENCIES} VERBATIM)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/native_resources.rcc
    COMMAND Qt6::rcc --binary ${CMAKE_CURRENT_SOURCE_DIR}/native/native_resources.qrc -o ${CMAKE_CURRENT_BINARY_DIR}/native_resources.rcc
    DEPENDS native/native_resources.qrc native/NativePageHost.qml VERBATIM)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/xovi.c ${CMAKE_CURRENT_BINARY_DIR}/xovi.h
    COMMAND ${Python3_EXECUTABLE} ${XOVI_SOURCE_DIR}/util/xovigen.py -o ${CMAKE_CURRENT_BINARY_DIR}/xovi.c -H ${CMAKE_CURRENT_BINARY_DIR}/xovi.h ${CMAKE_CURRENT_SOURCE_DIR}/native/reink-editor.xovi
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    DEPENDS native/reink-editor.xovi native/editor.qmd ${CMAKE_CURRENT_BINARY_DIR}/editor_panels.rcc ${CMAKE_CURRENT_BINARY_DIR}/native_resources.rcc ${XOVI_SOURCE_DIR}/util/xovigen.py VERBATIM)
  add_library(reink-editor-xovi MODULE native/entry.cpp ${CMAKE_CURRENT_BINARY_DIR}/xovi.c)
  target_include_directories(reink-editor-xovi PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(reink-editor-xovi PRIVATE repaper-editor-native)
  set_target_properties(reink-editor-xovi PROPERTIES PREFIX "" OUTPUT_NAME "reink-editor")
else()
  target_sources(repaper-editor-native PRIVATE editor_panels.qrc native/native_resources.qrc)
endif()
