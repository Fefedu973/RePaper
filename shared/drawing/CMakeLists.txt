cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RePaperNativeRegistry LANGUAGES CXX)
    include(CTest)
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
add_library(repaper-native-registry STATIC NativeObjectRegistry.cpp NativeObjectRegistry.h)
target_compile_features(repaper-native-registry PUBLIC cxx_std_17)
target_link_libraries(repaper-native-registry PUBLIC Qt6::Core)
target_include_directories(repaper-native-registry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
    add_executable(native-object-registry-tests tests/NativeObjectRegistryTest.cpp)
    set_target_properties(native-object-registry-tests PROPERTIES AUTOMOC ON)
    target_link_libraries(native-object-registry-tests PRIVATE repaper-native-registry Qt6::Test)
    add_test(NAME native-object-registry COMMAND native-object-registry-tests)
endif()
