cmake_minimum_required(VERSION 3.21)
project(repaper_stylus_input_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui GuiPrivate Quick Qml Test)
if(NOT FBCONTROLLER_SOURCE)
  message(FATAL_ERROR "Set FBCONTROLLER_SOURCE to the private patched source copy")
endif()
add_executable(stylus-input-tests StylusInputTests.cpp
  "${FBCONTROLLER_SOURCE}/src/qtfb/FBController.cpp"
  "${FBCONTROLLER_SOURCE}/src/qtfb/FBController.h")
target_include_directories(stylus-input-tests PRIVATE "${FBCONTROLLER_SOURCE}/src/qtfb")
target_compile_definitions(stylus-input-tests PRIVATE STYLUS_SOURCE="${FBCONTROLLER_SOURCE}")
target_link_libraries(stylus-input-tests PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::Quick Qt6::Qml Qt6::Test)
if(STYLUS_BASELINE)
  target_compile_definitions(stylus-input-tests PRIVATE STYLUS_BASELINE)
endif()
add_executable(linuxfb-stylus-client LinuxFbStylusClient.cpp)
target_include_directories(linuxfb-stylus-client PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../..")
target_link_libraries(linuxfb-stylus-client PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml)
