cmake_minimum_required(VERSION 3.16)
project(repaper_appload_device LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(APPLOAD_SOURCE "" CACHE PATH "Pinned AppLoad source containing src/qtfb/common.h")
if(NOT EXISTS "${APPLOAD_SOURCE}/src/qtfb/common.h")
    message(FATAL_ERROR "Set APPLOAD_SOURCE to the reviewed AppLoad source checkout")
endif()

add_executable(repaper-appload-launch launch.cpp)
add_executable(qt-linuxfb-refresh refresh.cpp)
target_include_directories(qt-linuxfb-refresh PRIVATE "${APPLOAD_SOURCE}/src/qtfb")
target_link_libraries(qt-linuxfb-refresh PRIVATE rt)
foreach(target repaper-appload-launch qt-linuxfb-refresh)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(${target} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
endforeach()
install(TARGETS repaper-appload-launch qt-linuxfb-refresh RUNTIME DESTINATION .)
