cmake_minimum_required(VERSION 3.21)
project(NativeAgendaTextTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Test)
add_executable(native-agenda-text-tests NativeAgendaTextTest.cpp ../../NativeAgendaText.cpp)
target_include_directories(native-agenda-text-tests PRIVATE ../..)
target_link_libraries(native-agenda-text-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
target_compile_options(native-agenda-text-tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME native-agenda-text COMMAND native-agenda-text-tests)
