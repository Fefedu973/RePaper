cmake_minimum_required(VERSION 3.21)
project(NativeAgendaHeaderTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Test)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
set(COMMON "${REPOSITORY}/extensions/editor-common")
add_executable(native-agenda-header-tests NativeAgendaHeaderTest.cpp ../../NativeAgendaHeader.cpp
  ${COMMON}/NativeObjectAccess.cpp ${COMMON}/NativeSelectionColor.cpp
  ${COMMON}/NativeLineFactory.cpp ${COMMON}/TargetProfile.cpp ${COMMON}/NativeStrokeSampling.cpp)
target_include_directories(native-agenda-header-tests PRIVATE ../.. ${COMMON} ${REPOSITORY}/apps/restencil/src)
target_link_libraries(native-agenda-header-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  target_compile_definitions(native-agenda-header-tests PRIVATE REPAPER_WITH_NATIVE_ABI=1)
  target_include_directories(native-agenda-header-tests PRIVATE ${REPOSITORY}/.tools/xovi-scene-assistant)
  target_sources(native-agenda-header-tests PRIVATE ${REPOSITORY}/.tools/xovi-scene-assistant/rm_Line.cpp)
endif()
target_compile_options(native-agenda-header-tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME native-agenda-header COMMAND native-agenda-header-tests)
