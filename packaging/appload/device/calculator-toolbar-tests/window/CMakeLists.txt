cmake_minimum_required(VERSION 3.21)
project(repaper_calculator_window_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Test)
if(NOT APPLOAD_SOURCE)
  message(FATAL_ERROR "Set APPLOAD_SOURCE to the private source copy after all AppLoad patches")
endif()
add_executable(calculator-window-tests CalculatorWindowTests.cpp "${APPLOAD_SOURCE}/src/Launcher.h")
target_include_directories(calculator-window-tests PRIVATE "${APPLOAD_SOURCE}/src")
target_compile_definitions(calculator-window-tests PRIVATE
  APPLOAD_QML_SOURCE="${APPLOAD_SOURCE}/resources/qml/appload.qml"
  APPLOAD_WINDOW_SOURCE="${APPLOAD_SOURCE}/resources/qml/window.qml")
target_link_libraries(calculator-window-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Test)
foreach(icon reload exit check keyboard)
  set_source_files_properties("${APPLOAD_SOURCE}/resources/icons/${icon}.svg" PROPERTIES QT_RESOURCE_ALIAS "icons/${icon}")
  list(APPEND test_icons "${APPLOAD_SOURCE}/resources/icons/${icon}.svg")
endforeach()
qt_add_resources(calculator-window-tests test_icons PREFIX "/appload" FILES ${test_icons})
enable_testing()
add_test(NAME calculator-window COMMAND calculator-window-tests)
set_tests_properties(calculator-window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
