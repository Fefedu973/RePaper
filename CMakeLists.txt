cmake_minimum_required(VERSION 3.21)
project(RePaper VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Network Sql Qml Quick QuickControls2 Concurrent)
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
endif()
find_package(OpenSSL REQUIRED)
add_subdirectory(shared)
add_subdirectory(bridge)
foreach(app remoodle reagenda restencil recalc reink)
  add_subdirectory(apps/${app})
endforeach()
find_path(PDFIUM_INCLUDE_DIR fpdfview.h)
set(_repdf_default OFF)
if(PDFIUM_INCLUDE_DIR)
  set(_repdf_default ON)
endif()
option(REPAPER_BUILD_REPDF "Build the AppLoad PDF reader (requires the PDFium public header)" ${_repdf_default})
if(REPAPER_BUILD_REPDF)
  add_subdirectory(apps/repdf)
endif()
option(REPAPER_BUILD_RMCHAT_PROBE "Build the archived experimental ChatGPT Web PC probe" OFF)
option(REPAPER_BUILD_RMCHAT "Build the archived RMChat application (explicit research opt-in)" OFF)
if(REPAPER_BUILD_RMCHAT OR REPAPER_BUILD_RMCHAT_PROBE)
  add_subdirectory(apps/rmchat)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME pc-handoff-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/test_pc_handoff.py)
  add_test(NAME emulator-profile-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/test_emulator_profiles.py)
endif()
