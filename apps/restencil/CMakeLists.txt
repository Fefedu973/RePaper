cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(reStencil VERSION 0.1.0 LANGUAGES CXX)
    include(CTest)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network Concurrent)
if(NOT TARGET repaper)
    find_package(OpenSSL REQUIRED)
    if(BUILD_TESTING)
        find_package(Qt6 REQUIRED COMPONENTS Test)
    endif()
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../shared ${CMAKE_CURRENT_BINARY_DIR}/shared)
endif()
add_executable(restencil src/main.cpp src/StencilCatalogue.cpp src/StencilCatalogue.h src/Geometry.cpp src/Geometry.h ../reink/src/InkCanvas.cpp ../reink/src/InkCanvas.h ../../extensions/restencil/src/ActivePageAdapter.cpp ../../extensions/restencil/src/ActivePageAdapter.h)
target_include_directories(restencil PRIVATE src ../reink/src ../../extensions/restencil/src)
target_link_libraries(restencil PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 repaper repaper_keyboard)
qt_add_resources(restencil restencil_qml PREFIX "/restencil" BASE qml FILES qml/Main.qml)
install(TARGETS restencil RUNTIME DESTINATION bin)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Qt6 6.2 QUIET COMPONENTS Test)
    if(TARGET Qt6::Test)
        add_executable(restencil_geometry_tests tests/GeometryTest.cpp src/Geometry.cpp src/Geometry.h ../../bridge/SceneWriter.cpp)
        target_include_directories(restencil_geometry_tests PRIVATE src ../../bridge)
        target_link_libraries(restencil_geometry_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
        add_test(NAME restencil_geometry COMMAND restencil_geometry_tests -platform offscreen)
        add_executable(restencil_ui_tests tests/StencilUiTest.cpp src/Geometry.cpp src/StencilCatalogue.cpp src/StencilCatalogue.h ../reink/src/InkCanvas.cpp ../reink/src/InkCanvas.h ../../extensions/restencil/src/ActivePageAdapter.cpp ../../extensions/restencil/src/ActivePageAdapter.h)
        target_include_directories(restencil_ui_tests PRIVATE src ../reink/src ../../extensions/restencil/src)
        target_link_libraries(restencil_ui_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Test repaper repaper_keyboard)
        qt_add_resources(restencil_ui_tests restencil_ui_qml PREFIX "/restencil" BASE qml FILES qml/Main.qml)
        add_test(NAME restencil_ui COMMAND restencil_ui_tests)
        set_tests_properties(restencil_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    endif()
endif()

include(${CMAKE_CURRENT_LIST_DIR}/../../shared/ui/PaperUi.cmake)
repaper_use_ui(restencil)
repaper_use_ui(restencil_ui_tests)
