cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(repdf VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  include(CTest)
endif()

find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Concurrent Network)
find_path(PDFIUM_INCLUDE_DIR fpdfview.h REQUIRED)
if(NOT TARGET repaper_keyboard)
  add_subdirectory(../../shared/keyboard ${CMAKE_CURRENT_BINARY_DIR}/keyboard)
endif()

add_library(repdf_core STATIC src/PdfView.cpp src/PdfView.h src/PdfLibrary.cpp src/PdfLibrary.h
  ../../shared/ui/AppLoadViewport.cpp ../../shared/ui/AppLoadViewport.h)
target_include_directories(repdf_core PUBLIC src PRIVATE ${PDFIUM_INCLUDE_DIR})
target_link_libraries(repdf_core PUBLIC Qt6::Core Qt6::Gui Qt6::Quick Qt6::Concurrent Qt6::Network)
add_executable(repdf src/main.cpp resources.qrc)
target_include_directories(repdf PRIVATE ../../shared/input)
target_link_libraries(repdf PRIVATE repdf_core repaper_keyboard Qt6::Qml Qt6::QuickControls2)
install(TARGETS repdf RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
  add_executable(repdf_view_tests tests/PdfViewTest.cpp)
  target_link_libraries(repdf_view_tests PRIVATE repdf_core Qt6::Test)
  add_test(NAME repdf_rendering COMMAND repdf_view_tests)
  set_tests_properties(repdf_rendering PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  add_executable(repdf_library_tests tests/PdfLibraryTest.cpp)
  target_link_libraries(repdf_library_tests PRIVATE repdf_core Qt6::Test)
  add_test(NAME repdf_file_selector COMMAND repdf_library_tests)
  set_tests_properties(repdf_file_selector PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS QuickTest)
    add_executable(repdf_qml_tests tests/QmlTestMain.cpp)
    target_link_libraries(repdf_qml_tests PRIVATE Qt6::QuickTest)
    add_test(NAME repdf_reader_ui COMMAND repdf_qml_tests
      -input ${CMAKE_CURRENT_SOURCE_DIR}/tests/qml
      -import ${CMAKE_CURRENT_SOURCE_DIR}/tests/qml/imports)
    set_tests_properties(repdf_reader_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  endif()
endif()

include(${CMAKE_CURRENT_LIST_DIR}/../../shared/ui/PaperUi.cmake)
repaper_use_ui(repdf)
repaper_use_ui(repdf_qml_tests)
