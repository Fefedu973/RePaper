  cmake_minimum_required(VERSION 3.21)
  project(reMoodle LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  include(CTest)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Network Sql Qml Quick QuickControls2 Concurrent)
  if(BUILD_TESTING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
  endif()
  find_package(OpenSSL REQUIRED)
  add_subdirectory(../../shared shared)

find_package(ZLIB REQUIRED)
add_library(moodle-domain STATIC MoodleController.cpp MoodleProtocol.cpp MoodleRelayClient.cpp PowerPointConverter.cpp PowerPointPackage.cpp third_party/qrcodegen.cpp)
target_include_directories(moodle-domain PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(moodle-domain PUBLIC repaper Qt6::Sql Qt6::Gui PRIVATE ZLIB::ZLIB)
add_executable(remoodle main.cpp resources.qrc)
target_include_directories(remoodle PRIVATE ../../shared/input)

install(TARGETS remoodle RUNTIME DESTINATION bin)
