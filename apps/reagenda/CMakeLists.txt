if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.21)
    project(reAgenda VERSION 0.1.0 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_AUTOMOC ON)
    include(CTest)
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Network Qml Quick QuickControls2)
if(NOT TARGET repaper)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Concurrent Test)
    find_package(OpenSSL REQUIRED)
    add_subdirectory(../../shared ${CMAKE_CURRENT_BINARY_DIR}/shared)
endif()
add_library(reagenda_core STATIC CalendarEngine.cpp CalendarEngine.h)
target_link_libraries(reagenda_core PUBLIC Qt6::Core)
target_include_directories(reagenda_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(reagenda main.cpp AgendaController.cpp AgendaController.h)
target_include_directories(reagenda PRIVATE ../../shared/input)
qt_add_resources(reagenda "reagenda_qml" PREFIX "/reagenda" FILES Main.qml)
target_link_libraries(reagenda PRIVATE reagenda_core Qt6::Gui Qt6::Network Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_link_libraries(reagenda PRIVATE repaper repaper_keyboard)
install(TARGETS reagenda RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
    qt_add_executable(reagenda_tests tests/CalendarEngineTest.cpp)
    target_link_libraries(reagenda_tests PRIVATE reagenda_core Qt6::Test)
    add_test(NAME reagenda_tests COMMAND reagenda_tests)
    if(NOT CMAKE_CROSSCOMPILING)
        qt_add_executable(reagenda_controller_tests tests/AgendaControllerTest.cpp AgendaController.cpp AgendaController.h)
        target_include_directories(reagenda_controller_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(reagenda_controller_tests PRIVATE reagenda_core repaper Qt6::Test)
        add_test(NAME reagenda_controller COMMAND reagenda_controller_tests)
        qt_add_executable(reagenda_ui_tests tests/AgendaUiTest.cpp AgendaController.cpp AgendaController.h)
        target_include_directories(reagenda_ui_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(reagenda_ui_tests PRIVATE reagenda_core repaper repaper_keyboard Qt6::Test)
        qt_add_resources(reagenda_ui_tests "reagenda_ui_qml" PREFIX "/reagenda" FILES Main.qml)
        add_test(NAME reagenda_ui COMMAND reagenda_ui_tests)
        set_tests_properties(reagenda_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
        qt_add_executable(reagenda_qtfb_probe tests/KeyboardQtfbProbe.cpp AgendaController.cpp AgendaController.h)
        target_include_directories(reagenda_qtfb_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(reagenda_qtfb_probe PRIVATE reagenda_core repaper repaper_keyboard)
        qt_add_resources(reagenda_qtfb_probe "reagenda_probe_qml" PREFIX "/reagenda" FILES Main.qml)
    endif()
endif()

include(${CMAKE_CURRENT_LIST_DIR}/../../shared/ui/PaperUi.cmake)
repaper_use_ui(reagenda)
repaper_use_ui(reagenda_ui_tests)
repaper_use_ui(reagenda_qtfb_probe)
