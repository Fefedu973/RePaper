cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(rmchat VERSION 0.1.0 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_AUTOMOC ON)
    include(CTest)
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Network)
find_package(OpenSSL REQUIRED)
add_library(rmchat_probe_support STATIC CoreProcess.cpp CoreProcess.h Credential.cpp Credential.h
    ../../shared/SecretStore.cpp ../../shared/SecretStore.h)
target_include_directories(rmchat_probe_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ../../shared)
target_link_libraries(rmchat_probe_support PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto)
add_executable(rmchat-probe main.cpp)
target_link_libraries(rmchat-probe PRIVATE rmchat_probe_support)
install(TARGETS rmchat-probe RUNTIME DESTINATION bin)
add_library(rmchat_controller STATIC ChatController.cpp ChatController.h)
target_link_libraries(rmchat_controller PUBLIC rmchat_probe_support)
option(RMCHAT_BUILD_UI "Build the native rmChat application" ON)
if(RMCHAT_BUILD_UI)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Gui Qml Quick QuickControls2)
    if(NOT TARGET repaper_keyboard)
        add_subdirectory(../../shared/keyboard ${CMAKE_CURRENT_BINARY_DIR}/keyboard)
    endif()
    add_subdirectory(rich)
    qt_add_executable(rmchat ui_main.cpp LocalFiles.cpp LocalFiles.h)
    target_include_directories(rmchat PRIVATE ../../shared/input)
    qt_add_resources(rmchat "rmchat_qml" PREFIX "/rmchat" FILES Main.qml)
    target_link_libraries(rmchat PRIVATE rmchat_controller rmchat_rich Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 repaper_keyboard)
    set(RMCHAT_CORE_BINARY "" CACHE FILEPATH "Prebuilt rmchat-core executable to package next to rmchat")
    if(RMCHAT_CORE_BINARY)
        if(NOT IS_ABSOLUTE "${RMCHAT_CORE_BINARY}" OR NOT EXISTS "${RMCHAT_CORE_BINARY}" OR IS_DIRECTORY "${RMCHAT_CORE_BINARY}")
            message(FATAL_ERROR "RMCHAT_CORE_BINARY must name an existing absolute executable path")
        endif()
        add_custom_target(rmchat_core_binary ALL
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RMCHAT_CORE_BINARY}" "$<TARGET_FILE_DIR:rmchat>/rmchat-core${CMAKE_EXECUTABLE_SUFFIX}"
            DEPENDS "${RMCHAT_CORE_BINARY}" VERBATIM)
        add_dependencies(rmchat rmchat_core_binary)
        install(PROGRAMS "${RMCHAT_CORE_BINARY}" DESTINATION bin RENAME "rmchat-core${CMAKE_EXECUTABLE_SUFFIX}")
    endif()
    install(TARGETS rmchat RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
    add_executable(rmchat_probe_tests tests/ProbeTest.cpp)
    target_link_libraries(rmchat_probe_tests PRIVATE rmchat_probe_support Qt6::Test)
    add_test(NAME rmchat_probe_tests COMMAND rmchat_probe_tests)
    if(NOT CMAKE_CROSSCOMPILING)
        add_executable(rmchat_controller_tests tests/ChatControllerTest.cpp)
        target_link_libraries(rmchat_controller_tests PRIVATE rmchat_controller Qt6::Test)
        if(RMCHAT_BUILD_UI)
            add_executable(rmchat_localfiles_tests tests/LocalFilesTest.cpp LocalFiles.cpp LocalFiles.h)
            target_include_directories(rmchat_localfiles_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
            target_link_libraries(rmchat_localfiles_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
            add_test(NAME rmchat_localfiles_tests COMMAND rmchat_localfiles_tests)
            target_sources(rmchat_controller_tests PRIVATE LocalFiles.cpp LocalFiles.h)
            target_compile_definitions(rmchat_controller_tests PRIVATE RMCHAT_TEST_UI=1)
            target_link_libraries(rmchat_controller_tests PRIVATE rmchat_rich Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 repaper_keyboard)
            qt_add_resources(rmchat_controller_tests "rmchat_test_qml" PREFIX "/rmchat" FILES Main.qml)
        endif()
        add_test(NAME rmchat_controller_tests COMMAND rmchat_controller_tests)
        set_tests_properties(rmchat_controller_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    endif()
endif()
