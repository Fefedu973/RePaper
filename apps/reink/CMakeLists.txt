cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(reInk VERSION 0.1.0 LANGUAGES CXX)
    include(CTest)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network Concurrent)
if(NOT TARGET repaper)
    find_package(OpenSSL REQUIRED)
    if(BUILD_TESTING)
        find_package(Qt6 REQUIRED COMPONENTS Test)
    endif()
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../shared ${CMAKE_CURRENT_BINARY_DIR}/shared)
endif()
add_executable(reink src/main.cpp src/InkCanvas.cpp src/InkCanvas.h ../restencil/src/Geometry.cpp ../restencil/src/Geometry.h ../../extensions/reink/editor_panels.qrc)
target_include_directories(reink PRIVATE ../restencil/src)
target_link_libraries(reink PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 repaper)
qt_add_resources(reink reink_qml PREFIX "/reink" BASE qml FILES qml/Main.qml)
install(TARGETS reink RUNTIME DESTINATION bin)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Qt6 6.2 QUIET COMPONENTS Test)
    if(TARGET Qt6::Test)
        add_executable(reink_canvas_tests tests/InkCanvasTest.cpp src/InkCanvas.cpp src/InkCanvas.h ../restencil/src/Geometry.cpp ../../extensions/reink/editor_panels.qrc)
        target_include_directories(reink_canvas_tests PRIVATE src ../restencil/src)
        target_link_libraries(reink_canvas_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Test repaper)
        qt_add_resources(reink_canvas_tests reink_test_qml PREFIX "/reink" BASE qml FILES qml/Main.qml)
        add_test(NAME reink_canvas COMMAND reink_canvas_tests -platform offscreen)
        add_executable(reink_graph_tests tests/DrawingGraphTest.cpp src/InkCanvas.cpp src/InkCanvas.h ../restencil/src/Geometry.cpp)
        target_include_directories(reink_graph_tests PRIVATE src ../restencil/src)
        target_link_libraries(reink_graph_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Test repaper)
        add_test(NAME reink_graph COMMAND reink_graph_tests -platform offscreen)
        add_executable(reink_alignment_tests tests/AlignmentGuideTest.cpp src/InkCanvas.cpp src/InkCanvas.h ../restencil/src/Geometry.cpp)
        target_include_directories(reink_alignment_tests PRIVATE src ../restencil/src)
        target_link_libraries(reink_alignment_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Test repaper)
        add_test(NAME reink_alignment COMMAND reink_alignment_tests -platform offscreen)
    endif()
endif()

include(${CMAKE_CURRENT_LIST_DIR}/../../shared/ui/PaperUi.cmake)
target_include_directories(reink PRIVATE ../../shared/ui)
